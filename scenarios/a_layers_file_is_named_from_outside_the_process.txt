# A layer's file is named by its absolute path, from outside the process: every
# `observe layer` reply carries `path`, it is absolute, it ends in `<layer>.wesl`, and
# two layers' paths differ only in the stem — which is what a caller on the socket
# names the file by.
#
# Run with:  just drive run scenarios/a_layers_file_is_named_from_outside_the_process.txt

observe log

fixed-delta 1/60

# `ridges` writes `base.wesl`, `height.wesl` and `moisture.wesl` into the project's
# shader directory, and bakes whole.
new 256 256 7 ridges
wait bake

# `path` is absolute and ends in `height.wesl`.
layer height
observe layer

# The same directory, a different stem.
layer base
observe layer

observe log
